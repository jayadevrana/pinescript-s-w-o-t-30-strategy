last_signal = 0
Buy_final = BuyCondition and (nz(last_signal[1]) == 0 or nz(last_signal[1]) == -1)

BuyHigh = ta.valuewhen(Buy_final, high, 0)

eShort = last_signal < 0 and close < ShortLow

LastSignal = 0
Buy = eBuy and (nz(LastSignal[1]) == 0 or nz(LastSignal[1]) == -1)
Short = eShort and (nz(LastSignal[1]) == 0 or nz(LastSignal[1]) == 1)

sellprice = ta.valuewhen(Short, low, 0)

sfirm = ta.crossunder(close, sellprice)

Exrem_Buy_Sell(BUY,SELL) =>

    buycond = BUY and BUY[1]==0
    sellcond = SELL and SELL[1]==0

    bullish = ta.barssince(buycond) < ta.barssince(sellcond)
    bearish = ta.barssince(sellcond) < ta.barssince(buycond)

    buym = bearish[1]and buycond
    sellm = bullish[1] and sellcond

    [buym,sellm]

[cbuy,ccb] = Exrem_Buy_Sell(bfirm,Buy)
